适用系统：RHEL / CentOS / Rocky / Fedora（dnf/yum） JDK 21 Spring Boot 3.5.16 MySQL 8.0 Vue 3 + Vite

一、架构与目录约定

浏览器 → Nginx(80) → 前端静态资源(/var/www/coal-web)
                    → /api 反代 → Spring Boot(8080) → MySQL(coal_sales)
/opt/coal-sales/           # 项目根（git clone）
├── coal-sales/            # 后端
├── coal-web/              # 前端
├── sql/                   # 数据库脚本
├── deploy.sh              # 首次部署脚本
├── update.sh              # 日常更新脚本
└── nginx/coal-sales.conf  # Nginx 配置

二、环境准备（一次性）

1. 基础工具

sudo dnf install -y git maven nginx

2. Node 20

curl -fsSL https://rpm.nodesource.com/setup_20.x | sudo bash -
sudo dnf install -y nodejs

3. MySQL

sudo dnf install -y mysql-server
sudo systemctl enable --now mysqld

4. 修改 MySQL root 密码

sudo grep 'temporary password' /var/log/mysqld.log   # 查看临时密码
mysql -u root -p
# 登录后执行：ALTER USER 'root'@'localhost' IDENTIFIED BY '你的新密码';

5. 验证

java -version && mvn -v && node -v && nginx -v && mysql --version

三、拉代码 + 初始化数据库

cd /opt
git clone https://github.com/你的用户名/coal-sales.git
cd coal-sales

# 建库 + 建应用账号
mysql -u root -p -e "CREATE DATABASE IF NOT EXISTS coal_sales DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci; CREATE USER 'coal'@'localhost' IDENTIFIED BY 'Coal@123456'; GRANT ALL ON coal_sales.* TO 'coal'@'localhost'; FLUSH PRIVILEGES;"

# 按依赖顺序导入 SQL
cd sql
for f in 01-permission 02-base-data 03-sales 04-stock 05-finance 06-audit 07-dashboard-permission-fix; do
  mysql -u root -p coal_sales < $f.sql
done

四、部署脚本

4.1 deploy.sh（首次部署）

#!/usr/bin/env bash
set -euo pipefail

DB_USER="coal"
DB_PASS="Coal@123456"
APP_HOME="/opt/coal-sales"
WEB_DIR="/var/www/coal-web"
JAR_NAME="coal-sales-0.0.1-SNAPSHOT.jar"

echo "==> 1/4 构建后端"
cd "$APP_HOME/coal-sales"
DB_USERNAME=$DB_USER DB_PASSWORD=$DB_PASS mvn clean package -DskipTests

echo "==> 2/4 构建前端"
cd "$APP_HOME/coal-web"
npm install
npm run build

echo "==> 3/4 部署前端静态文件"
sudo mkdir -p "$WEB_DIR"
sudo rm -rf "$WEB_DIR"/*
sudo cp -r dist/* "$WEB_DIR"/

echo "==> 4/4 启动后端"
pkill -f "$JAR_NAME" 2>/dev/null || true
cd "$APP_HOME/coal-sales"
nohup env DB_USERNAME=$DB_USER DB_PASSWORD=$DB_PASS \
  java -jar target/$JAR_NAME > "$APP_HOME/app.log" 2>&1 &
echo "后端 PID: $!"

echo "✅ 部署完成"

4.2 update.sh（日常更新）

#!/usr/bin/env bash
set -euo pipefail

DB_USER="coal"
DB_PASS="Coal@123456"
APP_HOME="/opt/coal-sales"
WEB_DIR="/var/www/coal-web"
JAR_NAME="coal-sales-0.0.1-SNAPSHOT.jar"

cd "$APP_HOME"
echo "==> 拉取最新代码"
git pull

if git diff --name-only HEAD@{1} HEAD | grep -q '^coal-sales/'; then
  echo "==> 后端有改动，重新打包并重启"
  cd coal-sales
  DB_USERNAME=$DB_USER DB_PASSWORD=$DB_PASS mvn clean package -DskipTests
  pkill -f "$JAR_NAME" 2>/dev/null || true
  nohup env DB_USERNAME=$DB_USER DB_PASSWORD=$DB_PASS \
    java -jar target/$JAR_NAME > "$APP_HOME/app.log" 2>&1 &
fi

if git diff --name-only HEAD@{1} HEAD | grep -q '^coal-web/'; then
  echo "==> 前端有改动，重新构建"
  cd "$APP_HOME/coal-web"
  npm run build
  sudo rm -rf "$WEB_DIR"/*
  sudo cp -r dist/* "$WEB_DIR"/
fi

echo "✅ 更新完成"

4.3 nginx/coal-sales.conf

server {
    listen 80;
    server_name _;

    root /var/www/coal-web;
    index index.html;

    location / {
        try_files $uri $uri/ /index.html;
    }

    location /api/ {
        proxy_pass http://127.0.0.1:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}

4.4 systemd 服务（推荐，替代 nohup）

文件：/etc/systemd/system/coal-sales.service

[Unit]
Description=Coal Sales Backend
After=network.target mysqld.service

[Service]
User=zengwei
Environment=DB_USERNAME=coal
Environment=DB_PASSWORD=Coal@123456
ExecStart=/usr/bin/java -jar /opt/coal-sales/coal-sales/target/coal-sales-0.0.1-SNAPSHOT.jar
WorkingDirectory=/opt/coal-sales/coal-sales
Restart=always
RestartSec=3

[Install]
WantedBy=multi-user.target
sudo systemctl daemon-reload
sudo systemctl enable --now coal-sales
sudo systemctl status coal-sales

五、执行步骤

# 1. 给脚本加权限
cd /opt/coal-sales
chmod +x deploy.sh update.sh

# 2. 首次部署
./deploy.sh

# 3. 配置 Nginx
sudo cp nginx/coal-sales.conf /etc/nginx/conf.d/
sudo nginx -t && sudo systemctl reload nginx

# 4. 开放防火墙（若启用 firewalld）
sudo firewall-cmd --permanent --add-service=http
sudo firewall-cmd --reload

# 5. 浏览器访问验证
# http://服务器IP

./update.sh

七、注意事项

#事项说明
1数据库密码Coal@123456 需改成实际密码，且不要提交进 git
2前端依赖coal-web/ 需先有 package.json 才能 npm install
3进程管理nohup 适合快速验证，生产建议用 systemd
4数据库变更写增量 SQL 单独执行，勿重跑全量脚本覆盖数据
5备份更新前先 mysqldump -u coal -p coal_sales > backup_$(date +%F).sql
